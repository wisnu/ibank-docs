### 1. Tabel Baru: `rekeningaktivitasnonfin`

Log aktivitas non-finansial nasabah (cek saldo, cek mutasi, cetak buku, dll.).

```sql
CREATE TABLE ibcore.rekeningaktivitasnonfin (
    id                  INTEGER GENERATED BY DEFAULT AS IDENTITY NOT NULL,
    nomor_rekening      VARCHAR(20)   NOT NULL,
    tanggal_aktivitas   TIMESTAMP     NOT NULL,
    kode_aktivitas      VARCHAR(20)   NOT NULL,  -- CEK_SALDO | CEK_MUTASI | LOGIN_MB | LOGIN_IB | LOGIN_ATM
    kode_channel        VARCHAR(10)   NULL,       -- ATM | MOBILE | IB | TELLER | API
    nomor_referensi     VARCHAR(50)   NULL,
    user_input          VARCHAR(20)   NULL,
    terminal_input      VARCHAR(19)   NULL,
    kode_cabang         VARCHAR(10)   NULL,
    tanggal_input       TIMESTAMP     DEFAULT CURRENT_TIMESTAMP,

    CONSTRAINT rekeningaktivitasnonfin_pkey PRIMARY KEY (id)
);

-- Index per rekening + tanggal (query audit)
CREATE INDEX idx_ran_norek_tgl
    ON ibcore.rekeningaktivitasnonfin
    (nomor_rekening, tanggal_aktivitas DESC);

-- Index untuk monitoring per jenis/kode_channel
CREATE INDEX idx_ran_jenis
    ON ibcore.rekeningaktivitasnonfin
    (kode_aktivitas, tanggal_aktivitas);

-- Index untuk purge / archival berkala
CREATE INDEX idx_ran_tglakt
    ON ibcore.rekeningaktivitasnonfin
    (tanggal_aktivitas);
```

### 10. Enhance `parametertransaksiumum`

Tambah kolom `tipe_exclude_aktivitas_nasabah` (pengganti semantik `is_exclude_aktivitas_nasabah`, boolean → multi-value), serta kolom `is_transaksi_sistem`, `allow_rekening_tidak_aktif`, `allow_rekening_dormant`.

| Kolom | Nilai | Keterangan |
|---|---|---|
| `tipe_exclude_aktivitas_nasabah` | `F` | Hitung sebagai aktivitas nasabah (default) |
| | `DC` | Exclude debit & kredit |
| | `D` | Exclude debit, kredit tetap dihitung |
| | `C` | Exclude kredit, debit tetap dihitung |
| `is_transaksi_sistem` | `T` / `F` | Tandai transaksi otomatis sistem |
| `allow_rekening_tidak_aktif` | `F`/`DC`/`D`/`C` | Izin transaksi pada rekening tidak aktif |
| `allow_rekening_dormant` | `F`/`DC`/`D`/`C` | Izin transaksi pada rekening dormant |

```sql
ALTER TABLE ibcore.parametertransaksiumum
    ADD COLUMN tipe_exclude_aktivitas_nasabah VARCHAR(2) DEFAULT 'F';

-- Migrasi nilai lama: 'T' → 'DC'
UPDATE ibcore.parametertransaksiumum
SET tipe_exclude_aktivitas_nasabah = 'DC'
WHERE kode_transaksi IN ('SD', 'PD', 'SC', 'SCD', 'SDP', 'SDZ', 'SI');

ALTER TABLE ibcore.parametertransaksiumum
    ADD COLUMN is_transaksi_sistem VARCHAR(1) DEFAULT 'F';

UPDATE ibcore.parametertransaksiumum
SET is_transaksi_sistem = 'T'
WHERE kode_transaksi IN ('SD', 'PD', 'SC', 'SCD', 'SDP', 'SDZ', 'SI');

ALTER TABLE ibcore.parametertransaksiumum
    ADD COLUMN allow_rekening_tidak_aktif VARCHAR(2) DEFAULT 'F';

ALTER TABLE ibcore.parametertransaksiumum
    ADD COLUMN allow_rekening_dormant VARCHAR(2) DEFAULT 'F';

UPDATE ibcore.parametertransaksiumum
SET allow_rekening_dormant = 'DC'
WHERE kode_transaksi IN ('SD', 'PD', 'SC', 'SCD', 'SDP', 'SDZ', 'SI');

-- Enum
DELETE FROM ibcore.enum_varchar WHERE enum_name = 'eTipeExcludeAktivitas';
INSERT INTO ibcore.enum_varchar (enum_name, enum_value, enum_description) VALUES ('eTipeExcludeAktivitas', 'F',  'Hitung sebagai aktivitas nasabah');
INSERT INTO ibcore.enum_varchar (enum_name, enum_value, enum_description) VALUES ('eTipeExcludeAktivitas', 'DC', 'Exclude dari aktivitas nasabah');
INSERT INTO ibcore.enum_varchar (enum_name, enum_value, enum_description) VALUES ('eTipeExcludeAktivitas', 'D',  'Exclude debit');
INSERT INTO ibcore.enum_varchar (enum_name, enum_value, enum_description) VALUES ('eTipeExcludeAktivitas', 'C',  'Exclude kredit');
```

### 11. Migrasi Data Awal (One-time)

#### Migrasi 1/3 — `tgl_transaksi_terakhir` dari `histtransaksi`

```sql
UPDATE ibcore.rekeningliabilitas rl
SET tgl_transaksi_terakhir = GREATEST(
    COALESCE(rl.tgl_transaksi_terakhir, '1900-01-01'::TIMESTAMP),
    (
        SELECT MAX(t.tanggal_transaksi)
        FROM   ibcore.histdetiltransaksi d
               INNER JOIN ibcore.histtransaksi t ON t.id_transaksi = d.id_transaksi
               LEFT  JOIN ibcore.parametertransaksiumum ptu ON ptu.kode_transaksi = t.kode_transaksi
        WHERE  d.nomor_rekening = rl.nomor_rekening
          AND  t.status_otorisasi = 1
          AND  COALESCE(t.is_reversed, 'F') = 'F'
          AND  COALESCE(ptu.tipe_exclude_aktivitas_nasabah, 'F') != 'DC'
    )
)
WHERE EXISTS (
    SELECT 1
    FROM   ibcore.histdetiltransaksi d
           INNER JOIN ibcore.histtransaksi t ON t.id_transaksi = d.id_transaksi
           LEFT  JOIN ibcore.parametertransaksiumum ptu ON ptu.kode_transaksi = t.kode_transaksi
    WHERE  d.nomor_rekening = rl.nomor_rekening
      AND  t.status_otorisasi = 1
      AND  COALESCE(t.is_reversed, 'F') = 'F'
      AND  COALESCE(ptu.tipe_exclude_aktivitas_nasabah, 'F') != 'DC'
);
```

#### Migrasi 2/3 — `tgl_transaksi_terakhir` dari `transaksi` (data H+0)

```sql
UPDATE ibcore.rekeningliabilitas rl
SET tgl_transaksi_terakhir = GREATEST(
    COALESCE(rl.tgl_transaksi_terakhir, '1900-01-01'::TIMESTAMP),
    (
        SELECT MAX(t.tanggal_transaksi)
        FROM   ibcore.detiltransaksi d
               INNER JOIN ibcore.transaksi t ON t.id_transaksi = d.id_transaksi
               LEFT  JOIN ibcore.parametertransaksiumum ptu ON ptu.kode_transaksi = t.kode_transaksi
        WHERE  d.nomor_rekening = rl.nomor_rekening
          AND  t.status_otorisasi = 1
          AND  COALESCE(t.is_reversed, 'F') = 'F'
          AND  COALESCE(ptu.tipe_exclude_aktivitas_nasabah, 'F') != 'DC'
    )
)
WHERE EXISTS (
    SELECT 1
    FROM   ibcore.detiltransaksi d
           INNER JOIN ibcore.transaksi t ON t.id_transaksi = d.id_transaksi
           LEFT  JOIN ibcore.parametertransaksiumum ptu ON ptu.kode_transaksi = t.kode_transaksi
    WHERE  d.nomor_rekening = rl.nomor_rekening
      AND  t.status_otorisasi = 1
      AND  COALESCE(t.is_reversed, 'F') = 'F'
      AND  COALESCE(ptu.tipe_exclude_aktivitas_nasabah, 'F') != 'DC'
);
```
